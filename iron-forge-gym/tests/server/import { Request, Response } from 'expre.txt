import { Request, Response } from 'express';
import WorkoutService from '../services/workoutService';

class WorkoutController {
    public async create(req: Request, res: Response) {
        try {
            const userId = req.body.userId || req.params.userId;
            const workout = await WorkoutService.createWorkout(userId, req.body);
            res.status(201).json(workout);
        } catch (err: any) {
            res.status(400).json({ message: 'Error creating workout', error: err?.message ?? String(err) });
        }
    }

    public async listByUser(req: Request, res: Response) {
        try {
            const userId = req.params.userId;
            const workouts = await WorkoutService.getWorkoutsByUser(userId);
            res.status(200).json(workouts);
        } catch (err: any) {
            res.status(500).json({ message: 'Error fetching workouts', error: err?.message ?? String(err) });
        }
    }

    public async get(req: Request, res: Response) {
        try {
            const workout = await WorkoutService.getWorkoutById(req.params.id);
            if (!workout) return res.status(404).json({ message: 'Workout not found' });
            res.status(200).json(workout);
        } catch (err: any) {
            res.status(500).json({ message: 'Error fetching workout', error: err?.message ?? String(err) });
        }
    }

    public async update(req: Request, res: Response) {
        try {
            const updated = await WorkoutService.updateWorkout(req.params.id, req.body);
            res.status(200).json(updated);
        } catch (err: any) {
            res.status(400).json({ message: 'Error updating workout', error: err?.message ?? String(err) });
        }
    }

    public async delete(req: Request, res: Response) {
        try {
            await WorkoutService.deleteWorkout(req.params.id);
            res.status(204).send();
        } catch (err: any) {
            res.status(500).json({ message: 'Error deleting workout', error: err?.message ?? String(err) });
        }
    }

    public async addExercise(req: Request, res: Response) {
        try {
            const updated = await WorkoutService.addExercise(req.params.id, req.body);
            res.status(200).json(updated);
        } catch (err: any) {
            res.status(400).json({ message: 'Error adding exercise', error: err?.message ?? String(err) });
        }
    }

    public async removeExercise(req: Request, res: Response) {
        try {
            const updated = await WorkoutService.removeExercise(req.params.id, req.params.exerciseId);
            res.status(200).json(updated);
        } catch (err: any) {
            res.status(400).json({ message: 'Error removing exercise', error: err?.message ?? String(err) });
        }
    }
}

export default new WorkoutController();
